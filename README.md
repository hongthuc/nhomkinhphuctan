    <script>
        /**
         * Xử lý fallback thông minh cho ảnh Google Drive:
         * 1. Thử link thumbnail chất lượng cao
         * 2. Nếu lỗi -> thử lh3.googleusercontent.com
         * 3. Nếu vẫn lỗi -> hiển thị ảnh placeholder sạch đẹp, không bao giờ gãy hình
         */
        function handleDriveImgError(img) {
            try {
                const step = parseInt(img.dataset.errStep || '0', 10);
                if (step === 0 && img.dataset.fallback) {
                    img.dataset.errStep = '1';
                    img.src = img.dataset.fallback;
                } else if (step <= 1 && img.dataset.placeholder) {
                    img.dataset.errStep = '2';
                    img.src = img.dataset.placeholder;
                }
            } catch (e) {
                // Giữ an toàn nếu có lỗi phát sinh
            }

        /* ================= 2 BANNER TỰ TRƯỢT + HỖ TRỢ VUỐT MOBILE ================= */
        let currentSlide = 0;
        const totalSlides = 2;
        let slideInterval = null;

        function updateSlider() {
            const track = document.getElementById('heroTrack');
            if (track) {
                track.style.transform = `translateX(-${currentSlide * 100}%)`;
            }
            document.querySelectorAll('.hero-dot').forEach((dot, idx) => {
                if (idx === currentSlide) {
                    dot.classList.add('w-6', 'bg-brand-gold');
                    dot.classList.remove('w-2.5', 'bg-white/50');
                } else {
                    dot.classList.remove('w-6', 'bg-brand-gold');
                    dot.classList.add('w-2.5', 'bg-white/50');
                }
            });
        }

        function goToSlide(index) {
            currentSlide = (index + totalSlides) % totalSlides;
            updateSlider();
            restartAutoSlide();
        }

        function restartAutoSlide() {
            if (slideInterval) clearInterval(slideInterval);
            slideInterval = setInterval(() => {
                currentSlide = (currentSlide + 1) % totalSlides;
                updateSlider();
            }, 5000);
        }

        document.getElementById('prevSlideBtn').addEventListener('click', () => goToSlide(currentSlide - 1));
        document.getElementById('nextSlideBtn').addEventListener('click', () => goToSlide(currentSlide + 1));

        // Hỗ trợ vuốt cảm ứng trên điện thoại cho Banner
        let touchStartX = 0;
        const heroSliderEl = document.getElementById('heroSlider');
        heroSliderEl.addEventListener('touchstart', (e) => {
            touchStartX = e.changedTouches[0].screenX;
        }, { passive: true });
        heroSliderEl.addEventListener('touchend', (e) => {
            const diff = e.changedTouches[0].screenX - touchStartX;
            if (Math.abs(diff) > 45) {
                if (diff < 0) goToSlide(currentSlide + 1);
                else goToSlide(currentSlide - 1);
            }
        }, { passive: true });
